ESP32 MicroPython 수신부 펌웨어
ESP32 보드에 아래 코드를 업로드하세요. 네오픽셀 LED(GPIO 14번 핀 연결, 4구)가 수신 값('1', '2', 'abc', '4')에 반응하여 색상을 제어합니다. 
# ESP32 MicroPython - Web BLE UART Neopixel Controller
import machine
import neopixel
import bluetooth
import time

# Nordic UART Service UUIDs (NUS)
NUS_UUID = bluetooth.UUID("6e400001-b5a3-f393-e0a9-e50e24dcca9e")
TX_UUID = bluetooth.UUID("6e400003-b5a3-f393-e0a9-e50e24dcca9e") # ESP32 -> Web
RX_UUID = bluetooth.UUID("6e400002-b5a3-f393-e0a9-e50e24dcca9e") # Web -> ESP32

# NeoPixel Setup (GPIO 14, 4 LEDs)
NUM_LEDS = 4
np = neopixel.NeoPixel(machine.Pin(14), NUM_LEDS)

def set_color(r, g, b):
    for i in range(NUM_LEDS):
        np[i] = (r, g, b)
    np.write()

# BLE Event Callback
def ble_irq(event, data):
    global conn_handle
    if event == 1: # Central Connect
        conn_handle = data[0]
        print("연결 수립 성공")
    elif event == 2: # Central Disconnect
        conn_handle = None
        print("연결 해제됨. 다시 스캔 대기 시작.")
        advertise()
    elif event == 3: # Central Write (Data received from client)
        handle, = data
        if handle == rx_handle:
            raw_data = ble.gatts_read(rx_handle)
            msg = raw_data.decode("utf-8").strip()
            print("수신된 명령:", msg)
            control_neopixel(msg)

def control_neopixel(command):
    if command == "1":
        set_color(255, 0, 0)      # 빨강 (NeoPixel Red)
    elif command == "2":
        set_color(0, 255, 0)      # 초록 (NeoPixel Green)
    elif command == "abc":
        set_color(0, 0, 255)      # 파랑 (NeoPixel Blue)
    elif command == "4":
        set_color(255, 255, 255)  # 흰색 (NeoPixel White)
    else:
        set_color(0, 0, 0)        # 소등 (NeoPixel Off)

# Initialize BLE
ble = bluetooth.BLE()
ble.active(True)
ble.irq(ble_irq)

# Service and Characteristics Setup
TX_CHAR = (TX_UUID, bluetooth.FLAG_NOTIFY)
RX_CHAR = (RX_UUID, bluetooth.FLAG_WRITE | bluetooth.FLAG_WRITE_NO_RESPONSE)
UART_SERVICE = (NUS_UUID, (TX_CHAR, RX_CHAR))
((tx_handle, rx_handle),) = ble.gatts_register_services((UART_SERVICE,))

def advertise():
    name = "ESP_LED_CTRL"
    payload = bytearray(b"\x02\x01\x06") # Flags
    payload += bytearray([len(name) + 1, 0x09]) + name.encode()
    
    # Advertise UART Service UUID (128-bit)
    uuid_bytes = bytes(reversed(NUS_UUID.data))
    payload += bytearray([17, 0x07]) + uuid_bytes
    
    ble.gap_advertise(100, payload)

advertise()
print("ESP32 BLE NeoPixel 가동 중...")
while True:
    time.sleep(1)
